cmake_minimum_required(VERSION 3.16)
project(Sonata LANGUAGES CXX)

include(CheckCXXSourceCompiles)
check_cxx_source_compiles("int main() { __uint128_t x=1; return __builtin_parityll(x); }" SONATA_WIDE_ARITHMETIC)
if(NOT SONATA_WIDE_ARITHMETIC)
  message(FATAL_ERROR "Use GCC or Clang with native 128-bit integer support.")
endif()

add_library(sonata INTERFACE)
target_include_directories(sonata INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(sonata INTERFACE cxx_std_17)
option(SONATA_EPOCH_DENSE_WEIGHTS "Use a dense table for checkpoint equality weights" OFF)
if(SONATA_EPOCH_DENSE_WEIGHTS)
  target_compile_definitions(sonata INTERFACE SONATA_EPOCH_DENSE_WEIGHTS)
endif()

include(CTest)
if(BUILD_TESTING)
  foreach(probe IN ITEMS handoff layer pending_child epoch epoch_envelope epoch_session)
    add_executable(sonata_${probe} tests/cpp/${probe}.cpp)
    target_link_libraries(sonata_${probe} PRIVATE sonata)
    add_test(NAME ${probe} COMMAND sonata_${probe})
  endforeach()
endif()

# Manual full-profile run. Building this target only compiles and links it.
add_executable(sonata_e2e EXCLUDE_FROM_ALL examples/end_to_end.cpp)
target_include_directories(sonata_e2e PRIVATE tests/cpp)
target_link_libraries(sonata_e2e PRIVATE sonata)
